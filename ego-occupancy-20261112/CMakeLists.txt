cmake_minimum_required(VERSION 3.16)
project(ego_occupancy VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
option(EGO_BUILD_RGBD "Build optional OpenCV RGB-D adapters" ON)
option(EGO_BUILD_DISPLAY "Build Pangolin occupancy display" OFF)
option(EGO_BUILD_ORB "Build experimental ORB RGB-D replay/live mapping entry" OFF)
option(EGO_BUILD_PLANNER "Build standalone planner core and offline tools" OFF)
set(ORB_SLAM3_ROOT "" CACHE PATH "Existing built ORB checkout")
set(EGO_ORB_CORE "" CACHE FILEPATH "Independent ORB library built with pose_bridge")
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(Threads REQUIRED)
include(CTest)
function(ego_warnings target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  if("${ARGV1}" STREQUAL "STRICT")
    target_compile_options(${target} PRIVATE -Werror)
  endif()
endfunction()
if(EGO_BUILD_DISPLAY AND NOT EGO_BUILD_RGBD)
  message(FATAL_ERROR "Display requires RGB-D/OpenCV adapters")
endif()
if(EGO_BUILD_ORB)
  if(NOT EGO_BUILD_RGBD OR NOT EXISTS "${ORB_SLAM3_ROOT}/include/System.h"
      OR NOT EXISTS "${EGO_ORB_CORE}")
    message(FATAL_ERROR "ORB entry requires RGB-D, existing ORB_SLAM3_ROOT and built EGO_ORB_CORE")
  endif()
  find_package(realsense2 CONFIG REQUIRED)
endif()
if(EGO_BUILD_RGBD)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d imgcodecs)
endif()
if(EGO_BUILD_DISPLAY OR EGO_BUILD_ORB)
  find_package(Pangolin REQUIRED)
endif()
add_subdirectory(src/orb_bridge)
add_subdirectory(src/plan_env)
add_subdirectory(apps/support)
if(EGO_BUILD_RGBD)
  add_subdirectory(src/sensor_io)
endif()
add_subdirectory(src/visualization)
if(EGO_BUILD_PLANNER)
  add_subdirectory(src/bspline)
  add_subdirectory(src/planner_common)
  add_subdirectory(src/path_searching)
  add_subdirectory(src/bspline_opt)
  add_subdirectory(src/plan_manage)
  include(apps/support/planning.cmake)
endif()
add_subdirectory(apps)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_subdirectory(tests)
endif()
